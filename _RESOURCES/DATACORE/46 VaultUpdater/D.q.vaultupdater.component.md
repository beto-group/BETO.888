            case 'support': return (<> <h3 style={STYLES.modalTitle}>Support the Developer</h3> <p style={STYLES.modalText}>If you find this tool helpful, please consider supporting its development.</p> <div style={STYLES.supportButtonContainer}> <button className="support-button" style={STYLES.supportButton} onClick={() => window.open('https://ko-fi.com/betogroup', '_blank')}> <div style={STYLES.supportButtonIcon} dangerouslySetInnerHTML={{ __html: ICONS.KOFI }}></div> <span style={STYLES.supportButtonText}>Ko-fi</span> </button> </div> <div style={{ ...STYLES.buttonGroup, marginTop: '20px' }}> <button style={{ ...STYLES.button, ...STYLES.secondaryButton }} onClick={handleCloseModal}>Maybe Later</button> </div> </>);
            case 'manifest_mismatch': const githubIssueUrl = `https://github.com/${GITHUB_OWNER}/${GITHUB_REPO}/issues/new?title=Manifest%20Mismatch%20for%20v${updateCheck.info.remoteVersion}&body=The%20updater%20detected%20that%20the%20manifest.json%20version%20is%20'${updateCheck.info.manifest.version}'%20while%20the%20changelog%20version%20is%20'${updateCheck.info.remoteVersion}'.`; return (<> <h3 style={{...STYLES.modalTitle, color: 'var(--text-warning)'}}>🤔 Houston, We Have a... Minor Discrepancy</h3> <p style={STYLES.modalText}>Looks like your buddy (that's me, the developer!) was in a bit of a rush. The "quick update" instructions don't quite match the new version.</p> <p style={{...STYLES.modalText, fontSize: '0.9em'}}><b>Changelog:</b> v{updateCheck.info.remoteVersion}<br/><b>Instructions:</b> v{updateCheck.info.manifest.version}</p> <p style={{...STYLES.modalText, fontSize: '0.9em'}}>You can still update perfectly fine, it'll just take the scenic route (a full download). No data will be lost.</p> <div style={STYLES.buttonGroup}> <button style={{ ...STYLES.button, ...STYLES.secondaryButton }} onClick={handleCloseModal}>Cancel</button> <button style={{ ...STYLES.button, ...STYLES.secondaryButton, fontSize: '11px' }} onClick={() => window.open(githubIssueUrl, '_blank')}>Gently Nudge the Dev</button> <button style={STYLES.button} onClick={handleStartForcedFullUpdate}>Take the Scenic Route</button> </div> </>);
            case 'full_update_required': const reason = updateCheck.info.manifest ? "You're a few updates behind." : "You're updating from an older version."; return (<> <h3 style={STYLES.modalTitle}>Full Update Required</h3> <p style={STYLES.modalText}>{reason}</p> <p style={STYLES.modalText}> To ensure everything is up-to-date, a full download of the vault is necessary. This is a one-time process to get you on the latest version. </p> <div style={STYLES.buttonGroup}> <button style={{ ...STYLES.button, ...STYLES.secondaryButton }} onClick={handleCloseModal}>Cancel</button> <button style={STYLES.button} onClick={handleStartForcedFullUpdate}>Proceed with Full Update</button> </div> </>);
            default: return null;
        }
    }

    const Banner = ({ onClick, onDevModeUnlock }) => { const DEV_CLICK_COUNT = 5; const DEV_CLICK_TIMEOUT_MS = 2000; const [devClickSequence, setDevClickSequence] = useState([]); const handleIconClick = (e) => { e.stopPropagation(); const now = Date.now(); const recentClicks = [...devClickSequence, now].filter(time => now - time < DEV_CLICK_TIMEOUT_MS); if (recentClicks.length >= DEV_CLICK_COUNT) { new Notice('Developer Mode Unlocked!', 3000); onDevModeUnlock(); setDevClickSequence([]); } else { setDevClickSequence(recentClicks); } }; if (updateCheck.status === 'checking') { return (<div style={{ ...STYLES.promoBanner, opacity: 0.7, cursor: 'default' }}><div style={STYLES.promoIconContainer}><AnimatedIcon svgString={ICONS.BELL} isActive={false} isInView={true} /></div><div style={STYLES.bannerTextContainer}><h3 style={STYLES.bannerTitle}>Checking for Updates...</h3><p style={STYLES.bannerText}>Please wait a moment.</p></div></div>); } if (updateCheck.info?.updateAvailable) { return (<div style={STYLES.promoBanner} className="promo-banner" onClick={onClick}><div style={STYLES.promoIconContainer} onClick={handleIconClick} title="Developer Access Trigger"><AnimatedIcon svgString={ICONS.BELL} isActive={true} isInView={true} /></div><div style={STYLES.bannerTextContainer}><h3 style={STYLES.bannerTitle}>{`Update Available: v${updateCheck.info.remoteVersion}`}</h3><p style={STYLES.bannerText}>{`You are on v${updateCheck.info.localVersion}. Click to upgrade.`}</p></div></div>); } return (<div style={STYLES.promoBanner} className="promo-banner" onClick={onClick}><div style={STYLES.promoIconContainer} onClick={handleIconClick} title="Developer Access Trigger"><AnimatedIcon svgString={ICONS.HEART} isActive={true} isInView={true} /></div><div style={STYLES.bannerTextContainer}><h3 style={STYLES.bannerTitle}>You Are Up-To-Date!</h3><p style={STYLES.bannerText}>Consider supporting the developer.</p></div></div>); };

    return (
        <div ref={containerRef}>
            <style>{STYLES.injectedStyles} {`.icon-button:hover { background-color: var(--background-modifier-border); color: var(--text-normal); }`}</style>
            {filePreview && <FilePreviewModal file={filePreview} onClose={() => setFilePreview(null)} />}

            {isFullTab ? (
                <div style={STYLES.fullTabWrapper} className={uniqueWrapperClass}>
                    <span style={STYLES.icon} className="subtle-icon" title="Exit Full Tab" onClick={handleExitFullTab}>&lt;/&gt;</span>
                    <button onClick={onReloadRequest} className="reload-button" style={STYLES.reloadButton} aria-label="Reload Component" title="Reload Component">
                        <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2.5" strokeLinecap="round" strokeLinejoin="round" style={{ transform: 'rotate(60deg)' }}><path d="M23 4v6h-6"></path><path d="M20.49 15a9 9 0 1 1-2.12-9.36L23 10"></path></svg>
                    </button>

                    {isDevMode ? (
                        <div style={{ ...STYLES.modalContent, position: 'relative', minWidth: '600px', maxWidth: '800px' }}>
                            <h3 style={STYLES.modalTitle}>Developer Tools</h3>
                            <p style={STYLES.modalText}>Generate the update manifest for the next release.</p>
                            
                            <div style={{ display: 'flex', alignItems: 'center', gap: '10px', marginBottom: '15px' }}>
                                <label htmlFor="branch-select" style={{ color: 'var(--text-muted)', fontVariant: 'small-caps' }}>Branch:</label>
                                <select 
                                    id="branch-select"
                                    value={selectedBranch} 
                                    onChange={e => setSelectedBranch(e.target.value)}
                                    disabled={branches.status !== 'success'}
                                    style={{ padding: '5px', borderRadius: '4px', backgroundColor: 'var(--background-secondary)', color: 'var(--text-normal)', border: '1px solid var(--background-modifier-border)' }}
                                >
                                    {branches.status === 'loading' && <option>Loading branches...</option>}
                                    {branches.status === 'error' && <option>Failed to load</option>}
                                    {branches.status === 'success' && branches.list.map(branchName => (
                                        <option key={branchName} value={branchName}>{branchName}</option>
                                    ))}
                                </select>
                            </div>
                            
                            <div style={{ ...STYLES.buttonGroup, marginTop: '10px' }}>
                                <button style={STYLES.button} onClick={() => handleGenerateManifest(selectedBranch)} disabled={manifestState.status === 'generating'}>
                                    {manifestState.status === 'generating' ? 'Generating...' : 'Generate Manifest'}
                                </button>
                            </div>
                            
                            {manifestState.status === 'generating' && (<div style={{ width: '100%', display: 'flex', flexDirection: 'column', alignItems: 'center' }}> <div style={STYLES.progressBarContainer}> <div style={{ ...STYLES.progressBar, width: `${(downloadProgress.total > 0 ? (downloadProgress.current / downloadProgress.total) * 100 : 0)}%` }} /> </div> <p style={STYLES.progressText}>Fetching remote files... ({downloadProgress.current}/{downloadProgress.total})</p> </div>)}
                            {manifestState.status === 'success' && manifestState.content && (<div style={{ width: '100%', marginTop: '20px' }}> <textarea readOnly style={{ ...STYLES.changelogContent, height: '250px', whiteSpace: 'pre', resize: 'vertical' }} value={manifestState.content} ></textarea> <div style={{ ...STYLES.buttonGroup, marginTop: '15px' }}> <button style={STYLES.button} onClick={handleCopyToClipboard}>Copy to Clipboard</button> </div> </div>)}
                            {manifestState.status === 'error' && (<p style={{ ...STYLES.modalText, color: 'var(--text-error)' }}>Error: {manifestState.error}</p>)}
                            <div style={{ ...STYLES.buttonGroup, position: 'absolute', bottom: '20px', right: '30px' }}>
                                <button style={{ ...STYLES.button, ...STYLES.secondaryButton }} onClick={handleExitDevMode}>Exit Dev Mode</button>
                            </div>
                        </div>
                    ) : (
                        isModalOpen && <div style={STYLES.modalOverlay} className="modal-fade-in" onClick={handleCloseModal}><div style={STYLES.modalContent} className="modal-scale-in" onClick={e => e.stopPropagation()}>{renderModalContent()}</div></div>
                    )}
                </div>
            ) : (
                <div style={{ display: 'flex', justifyContent: 'center', padding: '20px 0' }}>
                    <Banner onClick={handleEnterAndOpenModal} onDevModeUnlock={handleDevModeUnlock} />
                </div>
            )}
        </div>
    );
};
