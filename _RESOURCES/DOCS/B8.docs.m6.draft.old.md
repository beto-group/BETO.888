## 3D Scene Controls (Babylon.js)

**What it is**

Controls for creating and managing a 3D scene using a `<canvas>` element with the Babylon.js engine.

**Use when**

You need to render 3D models, create immersive game-like worlds, or display interactive 3D visualizations.

### Basic Scene Setup

Initializes the Babylon.js engine, a scene, a camera, and a basic light source. This is the foundation for any 3D component.

```jsx
// --- In a ViewComponent ---
const { useRef, useEffect } = dc;

function BasicScene() {
  const canvasRef = useRef(null);

  useEffect(() => {
    // Assumes Babylon.js is loaded (see Module 5 for loadScript)
    if (!canvasRef.current || !window.BABYLON) return;

    const engine = new BABYLON.Engine(canvasRef.current, true);
    const scene = new BABYLON.Scene(engine);
    
    const camera = new BABYLON.ArcRotateCamera("cam", -Math.PI / 2, Math.PI / 2.5, 10, BABYLON.Vector3.Zero(), scene);
    camera.attachControl(canvasRef.current, true);
    new BABYLON.HemisphericLight("light", new BABYLON.Vector3(0, 1, 0), scene);
    
    // Add a ground plane
    BABYLON.MeshBuilder.CreateGround("ground", { width: 6, height: 6 }, scene);
    
    engine.runRenderLoop(() => scene.render());

    return () => engine.dispose(); // Cleanup
  }, []);

  return <canvas ref={canvasRef} style={{ width: '100%', height: '100%' }} />;
}
```

### Loading a `.glb` Model

Loads a `.glb` 3D model file from the vault and adds it to the scene.

```jsx
async function loadGltfModel(scene, vaultPath) {
  // Assumes SceneLoader is loaded
  const modelUrl = dc.app.vault.adapter.getResourcePath(vaultPath);

  try {
    const result = await BABYLON.SceneLoader.ImportMeshAsync(
      null, "", modelUrl, scene, null, ".glb"
    );
    // The first mesh is often the root or main visible model
    return result.meshes[0];
  } catch (e) {
    console.error("Failed to load GLB model:", e);
    return null;
  }
}
```

### Video Texture

Applies a local video file from the vault as a "live" texture to a mesh.

```jsx
function applyVideoTexture(mesh, scene, vaultVideoPath) {
  const videoUrl = dc.app.vault.adapter.getResourcePath(vaultVideoPath);
  
  const videoTexture = new BABYLON.VideoTexture(
    "vidTex", videoUrl, scene, true, // true to generate mipmaps
    true, // true to auto-update
    BABYLON.Texture.TRILINEAR_SAMPLINGMODE,
    { autoPlay: true, loop: true, muted: true }
  );

  const material = new BABYLON.StandardMaterial("vidMat", scene);
  material.diffuseTexture = videoTexture;
  mesh.material = material;
}
```

## 2D Canvas & Animation Controls

**What it is**

Controls for creating 2D graphics and procedural animations using the HTML5 `<canvas>` element.

**Use when**

You need to create dynamic, animated backgrounds, generative art, or complex vector animations that CSS cannot easily handle.

### 2D Render Loop

Sets up a basic `requestAnimationFrame` loop for continuous drawing on a 2D canvas.

```jsx
// --- In a ViewComponent ---
const { useRef, useEffect } = dc;

function My2dAnimation() {
  const canvasRef = useRef(null);
  const animationFrameId = useRef(null);

  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas.getContext('2d');
    let frame = 0;

    const render = () => {
      frame++;
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      ctx.fillStyle = 'white';
      ctx.fillText(`Frame: ${frame}`, 10, 20);
      animationFrameId.current = requestAnimationFrame(render);
    };

    render();
    return () => cancelAnimationFrame(animationFrameId.current);
  }, []);

  return <canvas ref={canvasRef} width="300" height="150" />;
}
```

### Lottie Animation Player

Renders a complex vector animation from a `.json` file using the `<lottie-player>` web component.

```jsx
// --- In a ViewComponent ---
const { useState, useEffect } = dc;

function LottieViewer({ vaultJsonPath }) {
  const [lottieSrc, setLottieSrc] = useState(null);

  useEffect(() => {
    // See Module 5 for the advanced loadScript helper
    if (!window.customElements.get("lottie-player")) {
      loadScript("https://unpkg.com/@lottiefiles/lottie-player@latest/dist/lottie-player.js");
    }
    
    const url = dc.app.vault.adapter.getResourcePath(vaultJsonPath);
    setLottieSrc(url);
  }, [vaultJsonPath]);
  
  if (!lottieSrc) return <div>Loading...</div>;

  return (
    <lottie-player
      src={lottieSrc}
      background="transparent"
      speed="1"
      loop
      autoplay
    ></lottie-player>
  );
}
```

## Data Charting Controls (D3.js)

**What it is**

Controls for rendering data-driven, interactive charts and graphs using the D3.js library.

**Use when**

You need to build dashboards, visualize complex datasets with standard charts like bar or pie charts, or create custom, interactive visualizations.

### D3 Setup

Loads the D3.js library and creates an SVG container for the chart.

```jsx
// --- In a ViewComponent ---
const { useRef, useEffect, useState } = dc;

function MyChart() {
  const chartRef = useRef(null);
  const [isD3Ready, setIsD3Ready] = useState(!!window.d3);

  useEffect(() => {
    if (isD3Ready) return;
    // Assumes a loadScript helper is available
    loadScript("https://d3js.org/d3.v7.min.js", () => setIsD3Ready(true));
  }, [isD3Ready]);

  // D3 rendering logic will go here...
  
  // The ref points to the div that will contain the SVG.
  return <div ref={chartRef} style={{ width: '100%', height: '300px' }} />;
}
```

### Basic Bar Chart

A minimal example of rendering a simple bar chart from a data array.

```jsx
// --- Inside MyChart, within a useEffect that runs when isD3Ready changes ---
useEffect(() => {
  if (!isD3Ready || !chartRef.current) return;
  const d3 = window.d3;
  const data = [10, 45, 22, 60, 35];
  const width = 400, height = 200, margin = { top: 20, right: 20, bottom: 30, left: 40 };

  const svg = d3.select(chartRef.current).append("svg")
      .attr("width", width).attr("height", height);

  const xScale = d3.scaleBand().domain(d3.range(data.length)).range([margin.left, width - margin.right]).padding(0.1);
  const yScale = d3.scaleLinear().domain([0, d3.max(data)]).range([height - margin.bottom, margin.top]);

  svg.append("g").attr("fill", "steelblue")
    .selectAll("rect").data(data).join("rect")
    .attr("x", (d, i) => xScale(i))
    .attr("y", d => yScale(d))
    .attr("height", d => yScale(0) - yScale(d))
    .attr("width", xScale.bandwidth());
}, [isD3Ready]);
```
